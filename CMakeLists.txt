cmake_minimum_required(VERSION 3.16)
project(eurorack_patterns VERSION 0.2.1 LANGUAGES CXX)

add_library(eurorack_patterns INTERFACE)
add_library(eurorack_patterns::eurorack_patterns ALIAS eurorack_patterns)
target_include_directories(eurorack_patterns INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_compile_features(eurorack_patterns INTERFACE cxx_std_11)

set(ep_top_level OFF)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  set(ep_top_level ON)
endif()
option(EURORACK_PATTERNS_BUILD_TESTS "Build host tests and examples" ${ep_top_level})
option(EURORACK_PATTERNS_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(EURORACK_PATTERNS_BUILD_TESTS)
  enable_testing()
  function(ep_test target source)
    add_executable(${target} ${source})
    target_link_libraries(${target} PRIVATE eurorack_patterns)
    set_target_properties(${target} PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions -fno-rtti)
    endif()
    if(EURORACK_PATTERNS_SANITIZERS)
      if(MSVC)
        message(FATAL_ERROR "Sanitizer configuration requires GCC or Clang")
      endif()
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
    endif()
    add_test(NAME ${target} COMMAND ${target})
  endfunction()
  foreach(name presence events controls gestures timing transitions properties pickup pulses trigger_delay)
    ep_test(test_${name} tests/test_${name}.cpp)
    target_compile_definitions(test_${name} PRIVATE EP_TRACE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/traces")
  endforeach()
  ep_test(example_control_loop examples/control_loop.cpp)
  ep_test(example_event_paths examples/event_paths.cpp)
  ep_test(example_pickup_workflow examples/pickup_workflow.cpp)
  ep_test(test_shared_control tests/test_shared_control.cpp)
  ep_test(test_daisy_bench tests/test_daisy_bench.cpp)
  ep_test(test_daisy_experiment tests/test_daisy_experiment.cpp)
  ep_test(test_daisy_load tests/test_daisy_load.cpp)
  add_executable(daisy_replay examples/daisy/replay.cpp)
  target_link_libraries(daisy_replay PRIVATE eurorack_patterns)
  set_target_properties(daisy_replay PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  if(MSVC)
    target_compile_options(daisy_replay PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(daisy_replay PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions -fno-rtti)
  endif()
  if(EURORACK_PATTERNS_SANITIZERS)
    target_compile_options(daisy_replay PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
    target_link_options(daisy_replay PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
  endif()
  find_package(Python3 3.9 COMPONENTS Interpreter REQUIRED)
  add_test(NAME test_daisy_capture COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_daisy_capture.py $<TARGET_FILE:daisy_replay>)
  add_test(NAME test_daisy_measured COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_daisy_measured.py $<TARGET_FILE:daisy_replay>)
  add_test(NAME test_daisy_scope COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_daisy_scope.py)
  add_test(NAME test_daisy_fault_measured COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_daisy_fault_measured.py $<TARGET_FILE:daisy_replay>)
  add_test(NAME test_daisy_load_capture COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_daisy_load_capture.py $<TARGET_FILE:daisy_replay>)
  ep_test(example_behavior_traces examples/behavior_traces.cpp)
  # Each public header must compile with only its own includes.
  file(GLOB ep_headers CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/include/eurorack_patterns/*.h")
  foreach(header IN LISTS ep_headers)
    get_filename_component(stem "${header}" NAME_WE)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/header_${stem}.cpp" "#include <eurorack_patterns/${stem}.h>\nint main() { return 0; }\n")
    ep_test(header_${stem} "${CMAKE_CURRENT_BINARY_DIR}/header_${stem}.cpp")
  endforeach()
endif()

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS eurorack_patterns EXPORT eurorack_patternsTargets)
install(DIRECTORY include/ DESTINATION include)
install(FILES LICENSE ATTRIBUTION.md DESTINATION ${CMAKE_INSTALL_DATADIR}/eurorack_patterns)
install(EXPORT eurorack_patternsTargets NAMESPACE eurorack_patterns:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/eurorack_patterns)
configure_package_config_file(cmake/eurorack_patternsConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/eurorack_patternsConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/eurorack_patterns)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/eurorack_patternsConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/eurorack_patternsConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/eurorack_patternsConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/eurorack_patterns)
