cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED LIBDAISY_DIR OR NOT DEFINED EP_BUILD_ID)
  message(FATAL_ERROR "Use build.py to pin dependencies and record the build identity")
endif()
include(${LIBDAISY_DIR}/cmake/toolchains/autodetect.cmake)
project(patterns_daisy LANGUAGES C CXX ASM)
add_subdirectory(${LIBDAISY_DIR} ${CMAKE_BINARY_DIR}/libdaisy)
set(FIRMWARE_NAME patterns_daisy)
set(FIRMWARE_SOURCES main.cpp)
set(DAISY_STORAGE sram)
set(DAISY_GENERATE_BIN ON)
include(DaisyProject)
target_compile_definitions(patterns_daisy PRIVATE BOOT_APP)
target_include_directories(patterns_daisy PRIVATE ../../include ${CMAKE_BINARY_DIR})
target_compile_options(patterns_daisy PRIVATE -Wall -Wextra -Werror)
configure_file(build_identity.h.in build_identity.h @ONLY)
target_link_options(patterns_daisy PRIVATE "LINKER:-Map=${CMAKE_BINARY_DIR}/patterns_daisy.map")
