cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED LIBDAISY_DIR OR NOT DEFINED EP_BUILD_ID)
  message(FATAL_ERROR "Use build.py to pin dependencies and record the build identity")
endif()
include(${LIBDAISY_DIR}/cmake/toolchains/autodetect.cmake)
project(patterns_daisy LANGUAGES C CXX ASM)
add_subdirectory(${LIBDAISY_DIR} ${CMAKE_BINARY_DIR}/libdaisy)
if(EP_PULSE_GENERATOR AND EP_JACK_BENCH)
  message(FATAL_ERROR "Select only one firmware role")
endif()
if(EP_JACK_BENCH)
  set(FIRMWARE_NAME patterns_jack)
  set(FIRMWARE_SOURCES jack.cpp)
elseif(EP_PULSE_GENERATOR)
  set(FIRMWARE_NAME patterns_pulse_generator)
  set(FIRMWARE_SOURCES pulse_generator.cpp)
else()
  set(FIRMWARE_NAME patterns_daisy)
  set(FIRMWARE_SOURCES main.cpp)
endif()
set(DAISY_STORAGE sram)
set(DAISY_GENERATE_BIN ON)
include(DaisyProject)
target_compile_definitions(${FIRMWARE_NAME} PRIVATE BOOT_APP)
target_include_directories(${FIRMWARE_NAME} PRIVATE ../../include ${CMAKE_BINARY_DIR})
target_compile_options(${FIRMWARE_NAME} PRIVATE -Wall -Wextra -Werror)
configure_file(build_identity.h.in build_identity.h @ONLY)
target_link_options(${FIRMWARE_NAME} PRIVATE "LINKER:-Map=${CMAKE_BINARY_DIR}/${FIRMWARE_NAME}.map")
