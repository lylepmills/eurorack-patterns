cmake_minimum_required(VERSION 3.16)
if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
  set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
  message(FATAL_ERROR "Set PICO_SDK_PATH to a separate Pico SDK checkout")
endif()
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(patterns_pico LANGUAGES C CXX ASM)
pico_sdk_init()
set(EURORACK_PATTERNS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. patterns)
add_executable(patterns_pico main.cpp)
target_link_libraries(patterns_pico PRIVATE eurorack_patterns::eurorack_patterns pico_stdlib hardware_adc)
target_compile_options(patterns_pico PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Werror;-fno-exceptions;-fno-rtti>)
set_target_properties(patterns_pico PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
pico_enable_stdio_uart(patterns_pico 0)
pico_enable_stdio_usb(patterns_pico 0)
