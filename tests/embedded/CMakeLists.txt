cmake_minimum_required(VERSION 3.16)
project(patterns_arm_smoke LANGUAGES CXX)
set(EURORACK_PATTERNS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. patterns)
add_library(arm_smoke OBJECT smoke.cpp)
target_link_libraries(arm_smoke PRIVATE eurorack_patterns::eurorack_patterns)
target_compile_options(arm_smoke PRIVATE -mcpu=cortex-m0plus -mthumb -Os -Wall -Wextra -Wpedantic -Werror -fno-exceptions -fno-rtti)
set_target_properties(arm_smoke PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
# Also compile each header in isolation on the target architecture.
file(GLOB headers "../../include/eurorack_patterns/*.h")
foreach(header IN LISTS headers)
  get_filename_component(stem "${header}" NAME_WE)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${stem}.cpp" "#include <eurorack_patterns/${stem}.h>\n")
  target_sources(arm_smoke PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${stem}.cpp")
endforeach()
